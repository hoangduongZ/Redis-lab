Spring Boot + ZSet: "trượt thẻ trên bảng gỗ"

Không cần @Cacheable. Lần này dùng thẳng RedisTemplate để thao tác trực tiếp lên khe trượt. Hai dòng được tô sáng chính là toàn bộ "phép màu".

import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.data.redis.core.ZSetOperations;
import org.springframework.stereotype.Service;

import java.util.Set;

@Service
public class LeaderboardService {

    private final RedisTemplate<String, String> redisTemplate;

    public LeaderboardService(RedisTemplate<String, String> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    // 1. Hành động: Cập nhật điểm của người chơi
    public void addScore(String playerName, double score) {
        // Nếu playerName đã có, ZSet tự cộng/trừ điểm và tự trượt lên/xuống.
        redisTemplate.opsForZSet().add("game_leaderboard", playerName, score);
    }

    // 2. Hành động: Lấy Top 10 — gần như không tốn thời gian
    public Set<ZSetOperations.TypedTuple<String>> getTop10() {
        // reverseRangeWithScores: cao xuống thấp, vị trí 0 đến 9
        return redisTemplate.opsForZSet()
            .reverseRangeWithScores("game_leaderboard", 0, 9);
    }
}

Bài toán từng làm MySQL bốc khói giờ chỉ còn là một lệnh gọi hàm. Không ORDER BY, không quét bảng — dữ liệu vốn đã nằm đúng chỗ của nó.
